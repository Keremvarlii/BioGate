    const emailInput = document.getElementById('admin-email');
    const passwordInput = document.getElementById('admin-password');
    const loginMessage = document.getElementById('login-message');
    const loginBtn = document.getElementById('login-btn');

    loginForm.addEventListener('submit', (e) => {
        // Formun sayfayı yenilemesini engelliyoruz
        e.preventDefault();

        const email = emailInput.value;
        const password = passwordInput.value;

        // Butonu "Bekleniyor" moduna al
        loginBtn.textContent = "Doğrulanıyor...";
        loginBtn.disabled = true;
        loginMessage.textContent = "";

        // Backend API isteğini simüle ediyoruz (1 saniye bekleme)
        setTimeout(() => {
            // TEST İÇİN GEÇERLİ BİLGİLER
            const validEmail = "admin@biokilit.com";
            const validPassword = "admin";

            if (email === validEmail && password === validPassword) {
                // Şifre doğruysa
                loginMessage.style.color = "#10b981"; // Yeşil
                loginMessage.textContent = "Giriş başarılı, yönlendiriliyorsunuz...";
                
                // Yarım saniye sonra asıl panele yönlendir
                setTimeout(() => {
                    window.location.href = "adminAnaSayfa.html";
                }, 500);

            } else {
                // Şifre yanlışsa
                loginMessage.style.color = "#ef4444"; // Kırmızı

        }, 1000);
    });

import { ApiService } from './apiService.js';
import { UIManager } from './uiManager.js';

class App {
    constructor() {
        this.api = new ApiService();
        this.ui = new UIManager();
        
        // Gizli admin girişi için sayaç değişkenleri
        this.secretClickCount = 0;
        this.secretClickTimer = null;

        this.bindEvents();
        this.setupSecretAdminLogin();
    }

    bindEvents() {
        this.ui.submitBtn.addEventListener('click', () => this.handleVerification());
    }

    // 5 Tıklama Mantığını Kuran Fonksiyon
    setupSecretAdminLogin() {
        const logoTrigger = document.getElementById('secret-admin-trigger');
        
        if (logoTrigger) {
            logoTrigger.addEventListener('click', () => {
                this.secretClickCount++;
                
                // Önceki zamanlayıcıyı temizle
                clearTimeout(this.secretClickTimer);
                
                // 5 kez tıklandıysa sayfaya yönlendir
                if (this.secretClickCount >= 5) {
                    window.location.href = 'adminGiris.html';
                }

                // Eğer 1.5 saniye içinde yeni bir tıklama gelmezse sayacı sıfırla
                this.secretClickTimer = setTimeout(() => {
                    this.secretClickCount = 0;
                }, 1500);
            });
        }
    }

    async handleVerification() {
        const otpValue = this.ui.getOTPValue();

        if (otpValue.length !== 6) {
            this.ui.showError("Lütfen 6 haneli şifreyi eksiksiz giriniz.");
            return;
        }

        this.ui.showLoading();

        try {
            const response = await this.api.verifyOTP(otpValue);
            
            if (response.success) {
                this.ui.showSuccess("✓ Giriş Başarılı! Kapı Açılıyor...");
            } else {
                this.ui.showError("✗ Hatalı Şifre. Lütfen tekrar deneyin.");
            }
        } catch (error) {
            this.ui.showError("Sunucu bağlantı hatası oluştu.");
        }
    }
}

document.addEventListener('DOMContentLoaded', () => {
    new App();
});

export class UIManager {
    constructor() {
        this.inputs = document.querySelectorAll('.otp-inputs input');
        this.statusMessage = document.getElementById('status-message');
        this.submitBtn = document.getElementById('submit-btn');
        this.setupInputBehaviors();
    }

    setupInputBehaviors() {
        this.inputs.forEach((input, index) => {
            input.addEventListener('input', (e) => {
                e.target.value = e.target.value.replace(/[^0-9]/g, ''); // Sadece rakam
                if (e.target.value !== '' && index < this.inputs.length - 1) {
                    this.inputs[index + 1].disabled = false;
                    this.inputs[index + 1].focus();
                }
            });

            input.addEventListener('keydown', (e) => {
                if (e.key === 'Backspace' && e.target.value === '' && index > 0) {
                    this.inputs[index - 1].focus();
                    this.inputs[index - 1].value = '';
                }
            });
        });
    }

    getOTPValue() {
        let otp = '';
        this.inputs.forEach(input => otp += input.value);
        return otp;
    }

    showLoading() {
        this.statusMessage.textContent = "Doğrulanıyor...";
        this.statusMessage.className = "status-message";
        this.submitBtn.disabled = true;
    }

    showSuccess(message) {
        this.statusMessage.textContent = message;
        this.statusMessage.className = "status-message success";
        this.submitBtn.disabled = false;
    }

    showError(message) {
        this.statusMessage.textContent = message;
        this.statusMessage.className = "status-message error";
        this.submitBtn.disabled = false;
        this.resetInputs();
    }

    resetInputs() {
        this.inputs.forEach(input => {
            input.value = '';
            input.disabled = true;
        });
        this.inputs[0].disabled = false;
        this.inputs[0].focus();
    }
}
