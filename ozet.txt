const User = require('../models/User');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

// Admin Giriş İşlemi
exports.adminLogin = async (req, res) => {
    try {
        const { email, password } = req.body;

        // 1. Alanların doluluğunu kontrol et
        if (!email || !password) {
            return res.status(400).json({ 
                success: false, 
                message: 'Lütfen e-posta ve şifrenizi girin.' 
            });
        }

        // 2. Kullanıcıyı bul
        const user = await User.findOne({ email: email.toLowerCase().trim() });
        if (!user) {
            return res.status(401).json({ 
                success: false, 
                message: 'Geçersiz e-posta veya şifre.' 
            });
        }

        // 3. Rol kontrolü (Sadece admin girebilir)
        if (user.role !== 'admin') {
            return res.status(403).json({ 
                success: false, 
                message: 'Erişim reddedildi: Yönetici yetkiniz bulunmuyor.' 
            });
        }

        // 4. Şifre kontrolü
        const isMatch = await bcrypt.compare(password, user.password);
        if (!isMatch) {
            return res.status(401).json({ 
                success: false, 
                message: 'Geçersiz e-posta veya şifre.' 
            });
        }

        // 5. JWT Token Üretimi (8 saat geçerli)
        const token = jwt.sign(
            { id: user._id, role: user.role, email: user.email },
            process.env.JWT_SECRET,
            { expiresIn: '8h' }
        );

        res.status(200).json({
            success: true,
            message: 'Giriş başarılı!',
            token,
            user: {
                id: user._id,
                fullName: user.fullName,
                email: user.email,
                role: user.role
            }
        });

    } catch (error) {
        console.error('Login Hatası:', error);
        res.status(500).json({ 
            success: false, 
            message: 'Sunucu tarafında bir hata oluştu.' 
        });
    }
};

// 1. Şifre Üretme ve Gönderme İşlemi
exports.sendOtp = async (req, res) => {
    try {
        const { email } = req.body;

        if (!email) {
            return res.status(400).json({ success: false, message: 'Lütfen e-posta adresi girin.' });
        }

        // Güvenlik: Kullanıcı sistemde kayıtlı mı kontrol et
        const user = await User.findOne({ email: email.toLowerCase().trim() });
        if (!user) {
            return res.status(404).json({ success: false, message: 'Bu e-posta sistemde kayıtlı değil.' });
        }

        // 6 haneli rastgele OTP üret (100000 - 999999 arası)
        const otpCode = Math.floor(100000 + Math.random() * 900000).toString();

        // Spam'i önlemek için kullanıcının eski OTP kayıtlarını sil
        await Otp.deleteMany({ email: user.email });

        // Yeni kodu veritabanına kaydet (TTL index sayesinde 3 dk sonra silinecek)
        await Otp.create({
            email: user.email,
            otpCode: otpCode
        });

        // Mail içeriğini hazırla ve gönder
        const mailOptions = {
            from: `"BioGate Güvenlik Sistemi" <${process.env.EMAIL_USER}>`,
            to: user.email,
            subject: 'BioGate - Geçiş Doğrulama Kodunuz',
            html: `
                <div style="font-family: Arial, sans-serif; padding: 20px; color: #333;">
                    <h2 style="color: #111827;">BioGate Geçiş Doğrulaması</h2>
                    <p>Sisteme giriş yapmak için tek kullanımlık şifreniz oluşturulmuştur.</p>
                    <div style="background-color: #f3f4f6; padding: 15px; font-size: 28px; font-weight: bold; letter-spacing: 8px; text-align: center; border-radius: 8px; margin: 20px 0;">
                        ${otpCode}
                    </div>
                    <p style="color: #ef4444; font-size: 14px;">Bu kod <strong>3 dakika</strong> boyunca geçerlidir. Lütfen güvenlik personeliniz dahil kimseyle paylaşmayın.</p>
                </div>
            `
        };

        await transporter.sendMail(mailOptions);

        res.status(200).json({ success: true, message: 'OTP e-posta adresinize gönderildi.' });

    } catch (error) {
        console.error('OTP Gönderme Hatası:', error);
        res.status(500).json({ success: false, message: 'Kod gönderilirken bir hata oluştu.' });
    }
};

// 2. Gelen Şifreyi Doğrulama İşlemi
exports.verifyOtp = async (req, res) => {
    try {
        const { email, otpCode } = req.body;

        if (!email || !otpCode) {
            return res.status(400).json({ success: false, message: 'E-posta ve kod gereklidir.' });
        }

        // Veritabanında eşleşen aktif kodu bul
        const validOtp = await Otp.findOne({ email: email.toLowerCase().trim(), otpCode: otpCode });

        if (!validOtp) {
            return res.status(400).json({ success: false, message: 'Hatalı veya süresi dolmuş kod!' });
        }

        // Doğrulama başarılıysa kodu hemen sil (tek kullanımlık olması için)
        await Otp.deleteOne({ _id: validOtp._id });

        res.status(200).json({ success: true, message: 'Doğrulama başarılı, kapı açılıyor.' });

    } catch (error) {
        console.error('OTP Doğrulama Hatası:', error);
        res.status(500).json({ success: false, message: 'Doğrulama sırasında hata oluştu.' });
    }
};

const jwt = require('jsonwebtoken');

module.exports = (req, res, next) => {
    // 1. İstek başlığından (Header) token'ı al (Bearer <token> formatında gelir)
    const authHeader = req.header('Authorization');
    
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
        return res.status(401).json({ success: false, message: 'Erişim reddedildi: Token bulunamadı.' });
    }

    const token = authHeader.split(' ')[1];

    try {
        // 2. Token'ın gerçekliğini ve süresini kontrol et
        const decoded = jwt.verify(token, process.env.JWT_SECRET);
        
        // 3. Çözülen kullanıcı bilgilerini isteğin (req) içine yerleştir
        req.user = decoded;
        
        // 4. Yetki kontrolü: Sadece admin mi?
        if (req.user.role !== 'admin') {
            return res.status(403).json({ success: false, message: 'Erişim reddedildi: Yetkisiz rol.' });
        }

        // Geçiş izni ver, asıl controller'a devam et
        next();
    } catch (error) {
        return res.status(401).json({ success: false, message: 'Geçersiz veya süresi dolmuş token.' });
    }
};

require('dotenv').config();

const User = require('./models/User');

const seedAdmin = async () => {
    try {
        await mongoose.connect(process.env.MONGO_URI);
        console.log('✓ Veritabanına bağlanıldı.');

        // Admin daha önce eklenmiş mi kontrol et
        const existingAdmin = await User.findOne({ email: 'admin@biogate.com' });
        if (existingAdmin) {
            console.log('ℹ Admin kullanıcısı zaten mevcut (admin@biogate.com).');
            process.exit();
        }

        // Şifreyi hashle
        const salt = await bcrypt.genSalt(10);
        const hashedPassword = await bcrypt.hash('admin123', salt);

        // Yeni admin kullanıcısını oluştur
        const adminUser = new User({
            fullName: 'Sistem Yöneticisi',
            email: 'admin@biogate.com',
            password: hashedPassword,
            role: 'admin',
            isActive: true
        });

        await adminUser.save();
        console.log('✓ Varsayılan Admin başarıyla oluşturuldu!');
        console.log('  E-posta : admin@biogate.com');
        console.log('  Şifre   : admin123');

        process.exit();
    } catch (error) {
        console.error('✗ Hata oluştu:', error.message);
        process.exit(1);
    }
};

seedAdmin();

            body: JSON.stringify({ email: gercekMail })
        });

        }
        
        process.exit();

document.addEventListener('DOMContentLoaded', () => {
    const loginForm = document.getElementById('admin-login-form');
    const emailInput = document.getElementById('admin-email');
    const passwordInput = document.getElementById('admin-password');
    const loginMessage = document.getElementById('login-message');
    const loginBtn = document.getElementById('login-btn');

    loginForm.addEventListener('submit', async (e) => {
        // Formun sayfayı yenilemesini engelliyoruz
        e.preventDefault();

        const email = emailInput.value.trim();
        const password = passwordInput.value.trim();

        // Butonu "Bekleniyor" moduna al
        loginBtn.textContent = "Doğrulanıyor...";
        loginBtn.disabled = true;
        loginMessage.textContent = "";

        try {
            // Backend API'mize gerçek bir POST isteği atıyoruz
            const response = await fetch('http://127.0.0.1:5001/api/auth/admin-login', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({ email, password })
            });

            const data = await response.json();

            if (data.success) {
                // Şifre doğruysa ve JWT Token geldiyse
                loginMessage.style.color = "#10b981"; // Yeşil
                loginMessage.textContent = "Giriş başarılı, yönlendiriliyorsunuz...";

                // Güvenlik anahtarını (token) ve kullanıcı bilgisini tarayıcıya kaydet
                localStorage.setItem('adminToken', data.token);
                localStorage.setItem('adminUser', JSON.stringify(data.user));

                // Yarım saniye sonra asıl panele yönlendir
                setTimeout(() => {
                    window.location.href = "adminAnaSayfa.html";
                }, 500);

            } else {
                // Backend'den hata döndüyse (Yanlış şifre vb.)
                loginMessage.style.color = "#ef4444"; // Kırmızı
                loginMessage.textContent = data.message || "Hatalı e-posta veya şifre!";

                // Butonu eski haline getir
                loginBtn.textContent = "Sisteme Giriş Yap";
                loginBtn.disabled = false;
                passwordInput.value = ""; // Şifre kutusunu temizle
                passwordInput.focus();
            }
        } catch (error) {
            console.error('Bağlantı Hatası:', error);
            loginMessage.style.color = "#ef4444";
            loginMessage.textContent = "Sunucuya bağlanılamadı. Node.js sunucusunu başlattığınızdan emin olun.";

            loginBtn.textContent = "Sisteme Giriş Yap";
            loginBtn.disabled = false;
        }
    });
});

async function fetchLogs() {
    const tableBody = document.getElementById('log-table-body');
    const token = localStorage.getItem('adminToken');

    if (!token) return;

    // Yükleniyor mesajı
    tableBody.innerHTML = '<tr><td colspan="5" style="text-align: center;">Loglar yükleniyor...</td></tr>';

    try {
        // Port 5001 üzerinden, JWT Token ile güvenli istek atıyoruz
        const response = await fetch('http://localhost:5001/api/logs', {
            method: 'GET',
            headers: {
                'Authorization': `Bearer ${token}`,
                'Content-Type': 'application/json'
            }
        });

        const data = await response.json();

        if (data.success) {
            renderLogs(data.data);
        } else {
            tableBody.innerHTML = `<tr><td colspan="5" style="text-align: center; color: #ef4444;">${data.message}</td></tr>`;
        }
    } catch (error) {
        console.error('Bağlantı hatası:', error);
        tableBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: #ef4444;">Sunucuya bağlanılamadı. Node.js sunucusunu kontrol edin.</td></tr>';
    }
}

function renderLogs(logs) {
    const tableBody = document.getElementById('log-table-body');
    tableBody.innerHTML = ''; // Eski statik verileri tamamen temizle

    if (logs.length === 0) {
        tableBody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: #9ca3af;">Henüz sistemde hiç kayıtlı log bulunmuyor.</td></tr>';
        return;
    }

    // Gelen her bir log için tabloya yeni bir satır ekle
    logs.forEach(log => {
        // Duruma göre rozet rengini belirle
        let badgeClass = 'badge-success';
        if (log.status === 'warning') badgeClass = 'badge-warning';
        if (log.status === 'danger') badgeClass = 'badge-danger';

        // Tarih ve saati Türkçe formatta ayarla
        const dateObj = new Date(log.createdAt);
        const dateStr = dateObj.toLocaleDateString('tr-TR', { day: '2-digit', month: 'long', year: 'numeric' });
        const timeStr = dateObj.toLocaleTimeString('tr-TR', { hour: '2-digit', minute: '2-digit' });

        // Kanıt görüntüsü varsa link koy, yoksa '-' koy
        const evidenceHtml = log.evidenceImage 
            ? `<a href="${log.evidenceImage}" class="view-image-link" target="_blank">Görüntüyü İncele</a>` 
            : '-';

        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${dateStr}, ${timeStr}</td>
            <td>${log.userName || 'Bilinmiyor'}</td>
            <td>${log.eventType}</td>
            <td><span class="badge ${badgeClass}">${log.statusText}</span></td>
            <td>${evidenceHtml}</td>
        `;
        
        tableBody.appendChild(tr);
    });
}

import { ApiService } from './apiService.js';
import { UIManager } from './uiManager.js';

class App {
    constructor() {
        this.api = new ApiService();
        this.ui = new UIManager();
        
        // Gizli admin girişi için sayaç değişkenleri
        this.secretClickCount = 0;
        this.secretClickTimer = null;

        this.bindEvents();
        this.setupSecretAdminLogin();
    }

    bindEvents() {
        this.ui.submitBtn.addEventListener('click', () => this.handleVerification());
    }

    // 5 Tıklama Mantığını Kuran Fonksiyon
    setupSecretAdminLogin() {
        const logoTrigger = document.getElementById('secret-admin-trigger');
        
        if (logoTrigger) {
            logoTrigger.addEventListener('click', () => {
                this.secretClickCount++;
                
                // Önceki zamanlayıcıyı temizle
                clearTimeout(this.secretClickTimer);
                
                // 5 kez tıklandıysa sayfaya yönlendir
                if (this.secretClickCount >= 5) {
                    window.location.href = 'adminGiris.html';
                }

                // Eğer 1.5 saniye içinde yeni bir tıklama gelmezse sayacı sıfırla
                this.secretClickTimer = setTimeout(() => {
                    this.secretClickCount = 0;
                }, 1500);
            });
        }
    }

    async handleVerification() {
        const otpValue = this.ui.getOTPValue();

        if (otpValue.length !== 6) {
            this.ui.showError("Lütfen 6 haneli şifreyi eksiksiz giriniz.");
            return;
        }

        this.ui.showLoading();

        try {
            const response = await this.api.verifyOTP(otpValue);
            
            if (response.success) {
                this.ui.showSuccess("✓ Giriş Başarılı! Kapı Açılıyor...");
            } else {
                this.ui.showError("✗ Hatalı Şifre. Lütfen tekrar deneyin.");
            }
        } catch (error) {
            this.ui.showError("Sunucu bağlantı hatası oluştu.");
        }
    }
}

export class UIManager {
    constructor() {
        this.inputs = document.querySelectorAll('.otp-inputs input');
        this.statusMessage = document.getElementById('status-message');
        this.submitBtn = document.getElementById('submit-btn');
        this.setupInputBehaviors();
    }

    setupInputBehaviors() {
        this.inputs.forEach((input, index) => {
            input.addEventListener('input', (e) => {
                e.target.value = e.target.value.replace(/[^0-9]/g, ''); // Sadece rakam
                if (e.target.value !== '' && index < this.inputs.length - 1) {
                    this.inputs[index + 1].disabled = false;
                    this.inputs[index + 1].focus();
                }
            });

            input.addEventListener('keydown', (e) => {
                if (e.key === 'Backspace' && e.target.value === '' && index > 0) {
                    this.inputs[index - 1].focus();
                    this.inputs[index - 1].value = '';
                }
            });
        });
    }

    getOTPValue() {
        let otp = '';
        this.inputs.forEach(input => otp += input.value);
        return otp;
    }

    showLoading() {
        this.statusMessage.textContent = "Doğrulanıyor...";
        this.statusMessage.className = "status-message";
        this.submitBtn.disabled = true;
    }

    showSuccess(message) {
        this.statusMessage.textContent = message;
        this.statusMessage.className = "status-message success";
        this.submitBtn.disabled = false;
    }

    showError(message) {
        this.statusMessage.textContent = message;
        this.statusMessage.className = "status-message error";
        this.submitBtn.disabled = false;
        this.resetInputs();
    }

    resetInputs() {
        this.inputs.forEach(input => {
            input.value = '';
            input.disabled = true;
        });
        this.inputs[0].disabled = false;
        this.inputs[0].focus();
    }
}
